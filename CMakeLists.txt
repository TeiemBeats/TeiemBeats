cmake_minimum_required(VERSION 3.22)
project(TeiemTG VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/JUCE" CACHE PATH "JUCE 8.0.12 checkout")
add_subdirectory("${JUCE_PATH}" JUCE)
juce_add_binary_data(ConsoleAssets SOURCES Assets/console.png)
juce_add_plugin(TeiemTG COMPANY_NAME "TeiemBeats" BUNDLE_ID "com.teiembeats.tg" PLUGIN_MANUFACTURER_CODE Teim PLUGIN_CODE Tgmt FORMATS VST3 PRODUCT_NAME "Teiem TG Mastering" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Mastering)
target_sources(TeiemTG PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(TeiemTG PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TeiemTG PRIVATE ConsoleAssets juce::juce_audio_processors juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
add_executable(dsp_tests tests/dsp_tests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
enable_testing()
add_test(NAME mastering_dsp COMMAND dsp_tests)
juce_add_console_app(processor_tests PRODUCT_NAME "Processor tests")
target_sources(processor_tests PRIVATE tests/processor_tests.cpp)
target_include_directories(processor_tests PRIVATE Source)
target_link_libraries(processor_tests PRIVATE TeiemTG ConsoleAssets juce::juce_audio_utils)
add_test(NAME processor_state COMMAND processor_tests)
if(UNIX AND NOT APPLE)
 add_executable(factory_smoke tests/factory_smoke.cpp)
 target_include_directories(factory_smoke PRIVATE "${JUCE_PATH}/modules/juce_audio_processors_headless/format_types/VST3_SDK")
 target_link_libraries(factory_smoke PRIVATE dl)
 add_test(NAME vst3_factory COMMAND factory_smoke "${CMAKE_BINARY_DIR}/TeiemTG_artefacts/Release/VST3/Teiem TG Mastering.vst3/Contents/x86_64-linux/Teiem TG Mastering.so")
endif()
if(WIN32)
 add_test(NAME editor_render COMMAND processor_tests --ui)
endif()
